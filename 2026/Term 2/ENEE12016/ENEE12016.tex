	\section{Signals and Times Systems}
	\subsection{Signals:}
	A signal is a physical quantity that varies by something. The most common ones are time, space, and angle. Basically if it ca be measured or its effects measured, it can be though of as a signal. For example:
	\begin{itemize}
		\item Radio waves
		\item AC waveform
		\item Temperature
		\item Tide height
		\item Digital bit streams
	\end{itemize}

	\subsection{Sampling:}
	A discrete signal can be made by sampling a continuous signal of set time intervals. The sampled frequency must be twice the continuous frequency, as a minimum. The high the sampled frequency, the more accurate the output waveform.
	\subsection{Nyquist Theorem:}
	If a function $x(t)$ contains a highest frequency of BHz. For it too be completely determined sampling must be done at $\frac{1}{2B}$s.
	\subsection{System:}
	A group of regular items interacting to form a while. Such as, power amp, transformer, or laptop.
	\subsection{Signal Transformation:}
	\subsubsection{Amplitued Transform}
	Scaling the signal
	$$x(t)\rightarrow ax(t)$$
	\subsubsection{Amplitude Shift}
	Vertical shift
	$$x(t)\rightarrow x(t)+b$$
	\subsubsection{Time Reversal}
	A flip across the $y$-axis
	$$x(t)\rightarrow x(-t)$$
	\subsubsection{Time Scaling}
	Changing the frequency
	$$x(t)\rightarrow x(bt)$$
	\begin{itemize}
		\item If $b>1$: Compressing the signal
		\item Else $0<b<1$: Expanded the signal
	\end{itemize}
	\subsubsection{Time Shifting}
	Horizontal movement
	$$x(t)\rightarrow x(t-t_{0})$$
	\begin{itemize}
		\item If $t_{0}>0$: Shifts signal right
		\item Else $t_{0}<0$: Shifts signal left
	\end{itemize}
	\subsection{Even and Odd Signals:}
	Even signals are symmetric around the origin.
	$$x(t)=x(-t)$$
	Odd signals are asymmetric.
	$$x(t)=-x(-t)$$
	Any signal can be decomponsed into even and odd components.
	$$x_{e}(t)=\frac{1}{2}[x(t)+x(-t)]$$
	$$x_{o}(t)=\frac{1}{2}[x(t)-x(-t)]$$
	Sub $t=-t$, into even components.
	\begin{align*}
		x_{e}(-t)&=\frac{1}{2}[x(-t)+x(-(-t))]\\
		&=\frac{1}{2}[x(t)+x(-t)]\\
		&=x_{e}(t)\\
		&\therefore\text{Even}
	\end{align*}
	Sub $t=-t$, into odd components.
	\begin{align*}
		x_{o}(-t)&=\frac{1}{2}[x(-t)-x(-(-t))]\\
		&=\frac{1}{2}[-x(t)+x(-t)]\\
		&=-\frac{1}{2}[x(t)-x(-t)]\\
		&=-x_{o}(t)\\
		&\therefore\text{Odd}
	\end{align*}
	$$x(t)=x_{e}(t)+x_{o}(t)$$
	\subsection{Periodic Signals:}
	$$x(t)=x(t+T_{0}),\;T_{0}>0$$
	This is where $T_{0}$ is the period.
	\subsubsection{Complex Form}
	\begin{align*}
		x(t)&=e^{j\omega_{0}(t+T_{0})}\\
		&=e^{j\omega_{0}t}e^{j\omega_{0}T_{0}}
	\end{align*}
	\subsubsection{Fundamental Period}
	$$T_{0}=\frac{2\pi}{\omega_{0}}$$
	\subsection{General Complex Exponential:}
	$$C=|C|e^{j\phi},\;\;\;a=r+j\omega$$
	\begin{align*}
		x(t)&=|C|e^{j\phi(r+j\omega_{0})t}\\
		&=|C|e^{rt}e^{j\phi}e^{j\omega_{0}t}
		&=|C|e^{rt}\cos{(\omega_{0}t+\phi)}+j|C|e^{rt}\sin{(\omega_{0}t+\phi)}
	\end{align*}
	\subsection{Step Function:}
	Switching operations require abrupt changes in $v(t)$ and $t$. At the switching point, it is undefined.
	\begin{align*}
		Ku(t)=0,\;t<0\\
		Ku(t)=K,\;t>0\\
		Ku(t)=undf,\;t=0
	\end{align*}
	\subsection{Impulse Function:}
	This function has infinite magnitude and zero duration. No practical application contains this, as nothing last with zero time. It is used to define derivative at discontinuity point. Discrete signals can be saved to be made up of multiple impulse functions.
	$$x[n]=\sum^{k=\inf}_{k=-\inf}x[k]\delta[n-k]$$
	\subsection{Signal Energy and Power:}
	\subsubsection{Continuous}
	$$E=\int^{t_{2}}_{t_{1}}|x(t)|^{2}dt$$
	$$P=\frac{1}{t_{2}-t_{1}}\int^{t_{2}}_{t_{1}}|x(t)|^{2}dt$$
	$$E_{\infty}=\lim_{T\to\infty}\int^{T}_{-T}|x(t)|^{2}dt$$
	$$P_{\infty}=\lim_{T\to\infty}\frac{1}{2T}\int^{T}_{-T}|x(t)|^{2}dt$$
	\subsubsection{Discrete}
	$$E=\sum^{n_{2}}_{n_{1}}|x[n]|^{2}$$
	$$P=\frac{1}{n_{2}-n_{1}}\sum^{n_{2}}_{n_{1}}|x[n]|^{2}$$
	$$E_{\infty}=\lim_{N\to\infty}\sum^{N}_{-N}|x[n]|^{2}$$
	$$P_{\infty}=\lim_{N\to\infty}\frac{1}{2N+1}\sum^{N}_{-N}|x[n]|^{2}$$
	\subsubsection{Fundamentals}
	If energy over an infinite interval is finite, $E_{\infty}<\infty$, then $P_{av}$ is 0, $P_{\infty}=0$. Else $P_{\infty}>0\;\;\rightarrow\;\;E_{\infty}=\infty$. A few cases can have $E_{\infty}=\infty$ and $P_{\infty}=\infty$. One such one is $x(t)=t$.

	A system is casual if it only depends on present and past values.
	$$y(t)=5x(t)=6x^{2}(t-1)+y(t-1)$$

	A system is said to be invertible if the operations placed on it can be revered, giving a unique answer.

	\end{align*}

	\section{Linear Time Invariant Systems}
	\subsection{What are They?:}
	A linear system is one which obeys the laws of linearity and superposition:
	\begin{itemize}
		\item If $x$ is multiplied by $k$ making $kx$, $f(x)$ is also now $kf(x)$.
		\item When more than one factor is applied it is the sum of individual responses to each input.
	\end{itemize}
	\subsection{Time Invarience:}
	Behaviour and characteristic of the system is fixed over time. An example of this is an RC circuit characteristics will not change if the experiment is conducted today, tomorrow or yesterday. Mathematically, if the input is shifted by $T$, the output is also shifted by $T$.
	$$x(t)\Rightarrow y(t)$$
	$$x(t-T)\Rightarrow y(t-T)$$
	\subsection{Discrete Signals as Impulse Functions:}
	$$x[n]=\sum^{\infty}_{k=-\infty}x[k]\delta[n-k]$$
	For any given value of $n$, only 1 impulse in non-zero, and the rest are zero. Assume the response of an LTI system to unit impulse  $\delta[n]$ is $h[n]$. Thus $\delta[n-k]$ is $h[n-k]$.
	\subsection{Convolution of Discrete Signals:}
	The response of that system to signal $n[n]$ is superposition of a scaled and time shifted $h[n]$.
	$$y[n]=\sum^{\infty}_{k=-\infty}x[k]h[n-k]$$
	This is often called the convoluetion sum.
	\begin{align*}
		y[n]=&x[n]\otimes h[n]\\
		=&\sum^{\infty}_{k=-\infty}x[k]h[n-k]
	\end{align*}

	\subsection{Continuous Time Series Approximation:}
	This can be approximated as a series of pulse functions.
	\begin{itemize}
		\item Pulse located at integral multiples of pulse width ($\Delta t$)
		\item Pulse height is $x(\Delta t)$
	\end{itemize}
	Pulse of $t=\Delta t$.
	$$x(k\Delta t)[p(t-k\Delta t)\cdot\Delta t]=x(\Delta t)$$
	The signal can then be approximated as the sum of these signals.
	$$\hat{x}(t)=\sum^{\infty}_{k=-\infty}x(k\Delta t)[p(t-k\Delta t)\cdot\Delta t]$$
	As $\Delta t\to0$, $p(t)$ approaches an impulse function.
	$$x(t)=\int^{\infty}_{\infty}x(\tau)\delta(t-\tau)d\tau$$
	As $\Delta t\to0$, $k\Delta t\to\tau$, $\Delta t\to d\tau$ and $p(t-k\Delta t)\to\delta(t-\tau)$.

	If $h(t)$ is the response to the LTI system, $h_{p}(t)$ can be defined to be the response of a pulse system.

	\section{Laplace Transformation}
	\subsection{Why Laplace?:}
	\begin{itemize}
		\item Solving linear circuits frequently involves solving linear differential \\equations. These include transient analysis, AC analysis and transient or steady state system response.
		\item Finding time domain convolution
		\item Signals with many freqency components may be more apparent in a different domain
	\end{itemize}
	\subsection{Laplace Transform:}
	$$\Lap\{f(t)\}=F(s)=\int^{\infty}_{0}f(t)e^{-st}dt$$
	\begin{itemize}
		\item Convert a time domain function to frequency domain. $t\Rightarrow s,\;\text{where}\;s=\sigma+j\omega$
		\item Important Notes
		\begin{itemize}
			\item If intergrand does not converge there is no Laplace transform
			\item Only currently $t>0^{-}$
		\end{itemize}
	\end{itemize}
	\subsection{Conventions:}
	\begin{center}
	\begin{tabular}{ c c c }
		Time domain & & Frequency Domain\\
		$f(t)$ & $\Rightarrow$ & $F(s)$\\
		$g(t)$ & $\Rightarrow$ & $G(s)$
	\end{tabular}
	\end{center}
	Functional transform is only specific functions. Operation transofmr is the description of how mathematical operations are done using Laplace.
	\subsection{Integration by Parts:}
	$$\int^{\infty}_{0}uv'dt=[uv]^{\infty}_{0}-\int^{\infty}_{0}u'vdt$$
	Make sure that $u$ simplifies when derived and $v'$ simplifies when integrated. Laplace transform of 1:
	\begin{align*}
		\Lap\{f(t)\}&=F(s)=\int^{\infty}_{0}f(t)e^{-st}dt\\
		\Lap\{1\}&=\int^{\infty}_{0}e^{-st}dt\\
		&=[-\frac{1}{s}e^{-st}]^{\infty}_{0}\\
		&=[0]-[-\frac{1}{s}]\\
		&=\frac{1}{s}\\\\
		f(t)&\Leftrightarrow F(s)\\
		1&\Leftrightarrow \frac{1}{s}
	\end{align*}
	Laplace transform of $t$:
	\begin{align*}
		\Lap\{t\}&=F(s)=\int^{\infty}_{0}te^{-st}dt\\
		\text{Let}\;\;u&=t,\;\;v'=e^{-st}\\
		u'&=1,\;\;v=[-\frac{1}{s}e^{-st}]^{\infty}_{0}\\\\
		F(s)&=uv-\int u'vdt\\
		&=[t(-\frac{1}{s})]^{\infty}_{0}-\int^{\infty}_{0}-\frac{1}{s}e^{-st}dt\\
		&=0+\frac{1}{s}\int^{\infty}_{0}e^{-st}dt\\
		&=\frac{1}{s}[-\frac{1}{s}e^{-st}]^{\infty}_{0}\\
		&=-\frac{1}{s^{2}}[0-1]\\
		&=\frac{1}{s^{2}}\\\\
		t^{n}&\Leftrightarrow \frac{n!}{s^{n+1}}
	\end{align*}
	Laplace transform of $\delta(t)$:
	\begin{align*}
		\Lap\{(\delta(t)\}&=F(s)=\int^{\infty}_{0^{-}}\delta(t-0^{-})e^{-st}dt\\
		&=e^{-s\cdot 0}\\
		&=1\\\\
		\delta(t)&\Leftrightarrow 1\\
	\end{align*}
	In general
	$$\Lap\{\frac{d^{n}[\delta(t)]}{dt^{n}}\}=s^{n}$$
	\subsection{Operational Transforms:}
	\subsubsection{Multiplication by a Constant}
	\begin{align*}
		\Lap\{f(t)\}&=F(s)=\int^{\infty}_{0}f(t)e^{-st}dt\\
		\Lap\{Kf(t)\}&=\int^{\infty}_{0}Kf(t)e^{-st}dt\\\
		&=K\int^{\infty}_{0}f(t)e^{-st}dt\\
		&=KF(s)
	\end{align*}
	\subsubsection{Addition and Subtraction}
	\begin{align*}
		\Lap_{1}\{f_{1}(t)\}&=F_{1}(s)\\
		\Lap_{2}\{f_{2}(t)\}&=F_{2}(s)\\\\
		\Lap\{f_{1}(t)\pm f_{2}(t)\}&=\int^{\infty}_{0}(f_{1}(t)\pm f_{2}(t))e^{-st}dt\\
		&=\int^{\infty}_{0}f_{1}(t)e^{-st}dt\pm\int^{\infty}_{0}f_{2}(t)e^{-st}dt\\
		&=F_{1}(s)\pm F_{2}(s)
	\end{align*}
	\subsubsection{Differentiation}
	\begin{align*}
		\Lap\{\frac{df(t)}{dt}\}&=\int^{\infty}_{0^{-}}\frac{df(t)}{dt}e^{-st}dt\\
		\text{Let:}\;\;u'=\frac{df(t)}{dt}\;&\;v=e^{-st}\\
		u=f(t)\;&\;v'=-se^{-st}\\
		\Lap\{\frac{df(t)}{dt}\}&=uv-\int uv'\\
		&=[f(t)e^{-st}]^{\infty}_{0^{-}}-\int-sf(t)e^{-st}dt\\
		&=\int^{\infty}_{0^{-}}f(t)se^{-st}dt-f(0^{-})\\
		&=sF(s)-f(0^{-})
	\end{align*}
	In General:
	$$\Lap\{\frac{d^{n}f(t)}{dt^{n}}\}=s^{n}F(s)-s^{n-1}f(0^{-})-s^{n-2}\frac{df(0^{-}}{dt}-s^{n-3}\frac{d^{2}f(0^{-})}{dt^{2}}$$
	$$-\ldots-\frac{d^{n-1}f(0^{-})}{dt^{n-1}}$$
	\subsubsection{Integration}
	\begin{align*}
		\Lap\{\int^{t}_{0^{-}}f(x)dx\}&=\int^{\infty}_{0^{-}}[\int^{t}_{0^{-}}f(x)dx]e^{-st}dt\\
		\text{Let:}\;\;u'=e^{-st}\;&\;v=\int^{t}_{0^{-}}f(x)dx\\
		u=-\frac{1}{s}e^{-st}\;&\;v'=f(t)\\
		\Lap\{\int^{t}_{0^{-}}f(x)dx\}&=[-\frac{e^{-st}}{s}\int^{t}_{0^{-}}f(x)dx]^{\infty}_{0^{-}}-\int^{\infty}_{0^{-}}-f(t)\frac{e^{-st}}{s}dt\\
		\lim_{t\to\infty}&-\frac{e^{-st}}{s}\to0\\
		\lim_{t\to0^{-}}&\int^{t}_{0^{-}}f(x)dt\to0\\\
		\therefore\Lap\{\int^{t}_{0^-}\}&=\int^{\infty}_{0^{-}}f(t)\frac{e^{-st}}{s}dt\\
		&=\frac{1}{s}\int^{\infty}_{0^{-}}f(t)e^{-st}dt\\
		&=\frac{F(s)}{s}
	\end{align*}
	\subsubsection{Translation in Time Domain}
	$$\Lap\{f(t-a)u(t-a)\}=e^{-as}F(s)$$
	\subsubsection{Translation in Frequency Domain}
	$$\Lap\{e^{-at}f(t)\}=F(s+a)$$
	\subsubsection{Scale Change}
	$$\Lap\{f(at)\}=\frac{1}{a}F(\frac{s}{a})$$
	\subsubsection{Frequency Differentiation}
	$$\Lap\{tf(t)\}=-\frac{dF(s)}{ds}$$
	$$\Lap\{t^{n}f(t)\}=(-1)^{n}\frac{d^{n}F(s)}{ds^{n}}$$
	\subsection{Inverse Transform:}
	Once the time domain problem is solved using Laplace transformation, it needs to be converted back to time domain to understand what has happenend.
	$$F(s)=\frac{N(s)}{D(s)}=\frac{a_{n}s^{n}+a_{n-1}s^{n-1}+\ldots+a_{1}s^{1}+a_{0}s^{0}}{b_{m}s^{m}+b_{m-1}s^{m-1}+\ldots+b_{1}s^{1}+b_{0}s^{0}}$$
	As such, $F(s)$ is needed to be converted back into a format which contains Laplace transformations of known functions.

	\subsection{Circuit Elements in s-Domain:}
	Similar to analysis done on AC circuits in time domain. Using Laplace transforms, the time domain formulas can be converted to s-Domain. They key difference is that the initial conditions are taken into during the conversion.

	\subsubsection{Resistor}
	\begin{align*}
		v(t)&=Ri(t)\\
		\Lap\{v(t)\}&=V(s)=\int^{\infty}_{0}v(t)e^{-st}dt\\
		&=\int^{\infty}_{0}Ri(t)e^{-st}dt\\
		&=R\int^{\infty}_{0}i(t)e^{-st}dt\\
		\therefore V(s)&=RI(s)\\
		I(s)&=\frac{V(s)}{R}
	\end{align*}
	The units change in s-Domain to volt-seconds and amp-seconds.
	\subsubsection{Inductor}
	\begin{align*}
		v(t)&=L\frac{di(t)}{dt}\\
		\Lap\{v(t)\}&=V(s)=\int^{\infty}_{0}L\frac{di(t)}{dt}e^{-st}dt\\
		&=L\int^{\infty}_{0}\frac{di(t)}{dt}e^{-st}dt\\
		&=L(sI(s)-i(0^{-}))\\
		\therefore V(s)&=sLI(s)-LI_{0}\\
		I(s)&=\frac{V(s)}{sL}+\frac{I_{0}}{s}
	\end{align*}
	The $LI_{0}$ is the initial conditions of the inductor at $t=0^{-}$. $LI_{0}$ is a voltage sourse in series when finding voltage in s-Domain. $\frac{I_{0}}{s}$ is a current source in parrallel.
	\subsubsection{Capacitor}
	\begin{align*}
		i(t)&=C\frac{dv(t)}{dt}\\
		\Lap\{i(t)\}&=I(s)=\int^{\infty}_{0}C\frac{dv(t)}{dt}e^{-st}dt\\
		&=C\int^{\infty}_{0}\frac{dv(t)}{dt}e^{-st}dt\\
		&=C(sV(s)-v(0^{-}))\\\
		\therefore I(s)&=sCV(s)-CV_{0}\\
		V(s)&=\frac{I(s)}{sC}+\frac{V_{0}}{s}
	\end{align*}
	$V_{0}$ is the voltage at $t=0^{-}$.
	\subsection{Circuit Analysis:}
	All of Kirchoff's Laws (KCL and KVL) that applied in time domain, apply in s-Domain. The only important part is to have everything in one domain before analysis. The $R$, $sL$, and $\frac{1}{sC}$ are the s-Domain impedances for resistors, inductors and capacitors respectively.

	\subsection{Laplace vs. Fourier:}
	The fourier transform is a special case of the laplace transform. If all poles are on the left-side of the s-plane, the Fourier series converges. Then substitute $s=j\omega$.
	\begin{itemize}
		\item If $f(t)=0$ for $t\leq0$, $\For\{f(t)\}=\Lap\{f(t)\}|_{s=j\omega}$.
		\item Else $f(t)=0$ for $t\geq0$, $\For\{f(t)\}=\Lap\{f(t)\}|_{s=j\omega}$.
		\item Else $f(t)=a\;\forall\;t\in(-\infty,\infty)$:
			$$f(t)=f(t)^{+}+f(t)^{-}$$
			$$F(j\omega)=\Lap\{f(t)^{+}\}|_{s=j\omega}+\Lap\{f(-t)^{-}\}|_{s=-j\omega}$$
	\end{itemize}

		\item Have a finite number of discontinuities within any finite interval

		\item $\int^{\infty}_{-\infty}|f(t)|dt$ exists
	\end{enumerate}

		\therefore f(t)&=A\\

	\begin{align*}
		F(j\omega)&=\Re\{F(j\omega)\}+j\Im\{F(j\omega)\}\\
		\overline{F(j\omega)}&=\Re\{F(j\omega)\}-j\Im\{F(j\omega)\}\\
		&=\Re\{F(-j\omega)\}+j\Im\{F(-j\omega)\}
	\end{align*}
	If $f(t)$ is Real
	$$\Re\{F(j\omega)\}=\Re\{F(-j\omega)\}$$
	$$\Im\{F(j\omega)\}=-\Im\{F(-j\omega)\}$$
	Express $F(j\omega$ in polar form
	\begin{align*}
		F(j\omega)&=|F(j\omega)|e^{j\angle  F(j\omega)}\\
		|F(j\omega)|&=\sqrt{\Re\{F(j\omega)\}^{2}+\Im\{F(j\omega)\}^{2}}\\
		\angle F(j\omega)&=\arctan{\left(\frac{\Im\{F(j\omega)\}}{\Re\{F(j\omega)\}}\right)}
	\end{align*}
	If $f(t)$ is real and even:
	\begin{align*}
		F(j\omega)&=\int^{\infty}_{-\infty}f(t)e^{-j\omega t}dt\\
		\overline{F(j\omega)}&=F(-j\omega)\\
		&=\int^{\infty}_{-\infty}f(t)e^{j\omega t}dt\\
		\text{Sub}&\;t=-\tau\\
		F(-j\omega)&=-\int^{-\infty}_{\infty}f(-\tau)e^{-j\omega\tau}d\tau\\
		&=F(j\omega)\\
		&=\overline{F(j\omega)}
	\end{align*}
	If $f(t)$ is real and even, $F(j\omega)$ is real and even. Else $f(t)$ is real and odd, $F(j\omega)$ is imaginary and odd.
	\subsection{Duality Principle:}
	Between a Fourier Transform and its inverse are similar when applied in pairs. Generally:
	$$\For\{f(t)\}=F(\omega)$$
	$$\For\{F(\omega)\}=2\pi f(-\omega)$$
	\subsection{Parseval's Theorem:}
	If $f(t)$ and $F(j\omega)$ are Fourier pairs,
	$$\int^{\infty}_{-\infty}|f(t)|^{2}dt=\frac{1}{2\pi}\int^{\infty}_{-\infty}|F(j\omega)|^{2}d\omega$$
	The total energy within a signal can be found with either side of the above equation. As such $|F(j\omega)|^{2}$ is referred to the energy density spectrum. Note that if a finite amount of energy is calculated over an infinite interval, the power is $0$.

	\section{Frequency Selective Circuits}
	\subsection{Time vs. Frequency Domain:}
	Usually signals are most intuative in t-Domain (e.g. Audio, Power, or Control Systems). However, sometimes frequency-domain is intuative (e.g. Mixed Noise Signals). In this case the fouries transform shows how the signal energy is spread throughout the freqencies.
	\subsection{Signal Transmission:}
	To transmit a signal, all frequencies must be transmitted. This requiries a large amount of bandwidth, increasing the overall cost of the transmission. By finding where the energy is concentrated, only this part of the signal can be sent, reducing the bandwidth requirement.
	\subsection{Circuit Components and Frequency:}
	The impedance of certain circuit components varies with frequencies. These include capacitors, inductors and diodes.
	\subsection{Effects of Phase Changes:}
	\begin{itemize}
		\item Effects of magnitude change with many frequencies are easier to identify.
		\item Phase change can have a significant effect on output but harder to see.
	\end{itemize}
	\subsection{Frequency Response of an LTI System:}
	\begin{align*}
		H(j\omega)&=\frac{V_{out}(j\omega)}{V_{in}(j\omega)}\\
		&=|H(j\omega)|e^{j\angle H(j\omega)}\\
		&=|H(j\omega)|\angle H(j\omega)
	\end{align*}
	\subsection{Effects of Phase Response:}
	Linear Phase - Has the following response:
	\begin{align*}
		H(j\omega)&=e^{-j\omega t_{0}}\\
		|H(j\omega)|&=1\\
		\angle H(j\omega)&=-\omega t_{0}
	\end{align*}
	This system has a linear phase response, the output:
	\begin{align*}
		V_{o}(j\omega)&=H(j\omega)V_{i}(j\omega)\\
		&=e^{-j\omega t_{0}}V_{i}(j\omega)\\
		v_{o}(t)&=v_{i}(t-t_{0})\\
		\For\{f(t-t_{0})\}&=e^{-j\omega t_{0}}F(j\omega)
	\end{align*}
	\subsection{Bode Plot:}
	The frequency response of a system is understood better with a plot of it Magnitude and angle (in degrees) responses. This is usually done with a logarithmic scale.
	$$dB=20\log_{10}(|H(j\omega)|)$$
	\subsection{Frequency Selective Circuits:}
	System impedance for some circuits allow a high frequency to product a low amplitude. This allows certain frequencies to be passed or blocked. Some of these filter circuits include:
	\begin{itemize}
		\item Low Pass
		\item High Pass
		\item Band-pass
		\item Band-reject (Notch)
	\end{itemize}
	\subsection{Filter Categories - Ideal:}

	\subsubsection{Low Pass}
	\begin{center}
		\includegraphics[scale=0.25]{passive-low-pass.png}
	\end{center}
	Using voltage divider:
	\begin{align*}
		H(j\omega)&=\frac{V_{o}(j\omega)}{V_{i}(j\omega)}\\
		&=\frac{R}{R+j\omega L}\\
		&=\frac{\frac{R}{L}}{j\omega+\frac{R}{L}}\\
		|H(j\omega)|&=\frac{\frac{R}{L}}{\sqrt{\omega^{2}+\left(\frac{R}{L}\right)^{2}}}\\
		\angle H(j\omega)&=-\arctan\left(\frac{\omega L}{R}\right)
	\end{align*}
	Cut off Frequency defines the frequency where the average power is $\frac{1}{2}$ of Max:
	\begin{align*}
		P_{av}&=\frac{1}{2}\frac{V_{o}^{2}}{R}\\
		P_{av(max)}&=\frac{1}{2}\frac{V_{o(max)}^{2}}{R}\\
		P_{av(\omega_{c})}&=\frac{1}{2}\frac{V_{o(\omega_{c})}^{2}}{R}\\
		V_{o(\omega_{c})}^{2}&=\frac{1}{2}V_{o(max)}^{2}\\
		|H(j\omega_{c})|^{2}V_{i}^{2}&=\frac{1}{2}|H(j\omega)_{max}|^{2}V_{i}^{2}\\
		|H(j\omega_{c})|&=\frac{1}{\sqrt{2}}|H(j\omega)_{max}|
	\end{align*}
	Calculating $H(j\omega)_{max}$ is at $\omega=0$
	$$|H(j\omega)_{max}|=\frac{\frac{R}{L}}{\sqrt{\frac{R}{L}^{2}}}=1$$
	Cut off it at $H(j\omega_{c})=\frac{1}{\sqrt{2}}$
	\begin{align*}
		\frac{\frac{R}{L}}{\sqrt{\omega_{c}^{2}+\left(\frac{R}{L}\right)^{2}}}&=\frac{1}{\sqrt{2}}\\
		\omega_{c}^{2}+\left(\frac{R}{L}\right)^{2}&=2\left(\frac{R}{L}\right)^{2}\\
		\omega_{c}&=\frac{R}{L}
	\end{align*}
	The transfer function is best done in s-Domain, even if designed in frequency or t-Domain.
	\pagebreak
	$$\text{Sub}\;j\omega=s$$
	\begin{align*}
		H(j\omega)&=\frac{\frac{R}{L}}{j\omega+\frac{R}{L}}\\
		H(s)&=\frac{\frac{R}{L}}{s+\frac{R}{L}}\\
		\text{Since}\;\frac{R}{L}&=\omega_{c}\\
		H(s)&=\frac{\omega_{c}}{s+\omega_{c}}
	\end{align*}
	Cut off magnitude  is dB is 0 since the maximum magnitude is 1. At cutoff magnitude is $\frac{1}{\sqrt{2}}$, the dB is -3. In Bode plots, the cut off is identified as magnitude is 3dB below maximum magnitude.\\
	Rolloff is the slope of the magnitude response as it drops after the cut off frequency. Usually this is expressed in dB/decade.
	$$|H(j\omega)|=\frac{\omega_{c}}{\sqrt{\omega^{2}+\omega_{c}^{2}}}$$
	$$|H(j10\omega)|=\frac{\omega_{c}}{\sqrt{10^{2}\omega^{2}+\omega_{c}^{2}}}$$
	For frequencies where $\omega\gg\omega_{c}$:
	$$|H(j\omega)|=\frac{\omega_{c}}{\omega}$$
	$$|H(j10\omega)|=\frac{\omega_{c}}{10\omega}$$
	Converting the different to dB
	\begin{align*}
		&20\log_{10}\left(\frac{\omega_{c}}{\omega}\right)-20\log_{10}\left(\frac{\omega_{c}}{10\omega}\right)\\
		=&20\log_{10}(10)\\
		=&20\text{dB/decade}
	\end{align*}
	\subsubsection{High Pass Filter}
	\begin{center}
		\includegraphics[scale=0.5]{passive-high-pass.png}
	\end{center}
	\begin{align*}
		H(j\omega)&=\frac{V_{o}(j\omega)}{V_{i}(j\omega)}\\
		&=\frac{R}{R+\frac{1}{j\omega C}}\\
		&=\frac{j\omega}{j\omega+\frac{1}{RC}}
	\end{align*}
	Magnitude Response:
	$$|H(j\omega)|=\frac{\omega}{\sqrt{\omega^{2}+\left(\frac{1}{RC}\right)^{2}}}$$
	Phase Response:
	$$\angle H(j\omega)=\arctan\left(\frac{1}{\omega RC}\right)=90^{\circ}-\arctan(\omega RC)$$
	Cut off Frequency and Transfer Function:
	\begin{align*}
		H(j\omega)_{max}&=1\\
		\text{Then}\;H(j\omega_{c})&=\frac{1}{\sqrt{2}}\\
		\frac{\omega}{\sqrt{\omega^{2}+\left(\frac{1}{RC}\right)^{2}}}&=\frac{1}{\sqrt{2}}\\
		\omega_{c}^{2}+\left(\frac{1}{RC}\right)^{2}&=2\omega_{c}^{2}\\
		\omega_{c}&=\frac{1}{RC}\\\\
		H(j\omega)&=\frac{j\omega}{j\omega+\frac{1}{RC}}\\
		H(s)&=\frac{s}{s+\omega_{c}}
	\end{align*}
	\subsubsection{Band-pass Filter}
	\begin{center}
		\includegraphics[scale=0.5]{passive-bandpass-filter.png}
	\end{center}
	\begin{align*}
		H(j\omega)&=\frac{V_{o}(j\omega)}{V_{i}(j\omega)}\\
		&=\frac{R}{R+j\omega L+\frac{1}{j\omega C}}\\
		&=\frac{j\omega\frac{R}{L}}{-\omega^{2}+\frac{1}{LC}+j\omega\frac{R}{L}}
	\end{align*}
	Magnitude Response:
	$$h(j\omega)|=\frac{\omega\frac{R}{L}}{\sqrt{\left(\frac{1}{LC}-\omega^{2}\right)^{2}+\left(\frac{\omega R}{L}\right)^{2}}}$$
	Phase Response:
	$$\angle H(j\omega)=\arctan\left(\frac{\frac{1}{LC}-\omega^{2}}{\frac{\omega R}{L}}\right)=90^{\circ}-\arctan\left(\frac{\frac{\omega R}{L}}{\frac{1}{LC}-\omega^{2}}\right)$$
	\begin{center}
		\includegraphics[scale=0.2]{passive-bandpass-cutoff.png}
	\end{center}
	\subsubsection{Band-pass Characteristics}
	At the centre frequency, the capacitor and inductor impendance cancel each other out.
	$$H(j\omega_{0})=\frac{R}{R+j\omega_{0}L+\frac{1}{j\omega_{0}C}}=\frac{R}{R}=1$$
	When $|H(j\omega)_{max}|=1$:
	$$j\omega_{0}L-\frac{j}{\omega_{0}C}=0$$
	$$\omega_{0}=\sqrt{\frac{1}{LC}}$$
	Cut off Frequency:
	\begin{align*}
		|H(j\omega_{c})|&=\frac{\frac{\omega_{c}R}{L}}{\sqrt{\left(\frac{1}{LC}-\omega_{c}^{2}\right)+\left(\frac{\omega_{c}R}{L}\right)^{2}}}\\
		&=\frac{1}{\sqrt{2}}\\\\
		\therefore\omega_{c}^{2}L\pm\omega_{c}R-\frac{1}{C}&=0\\
		\omega_{c1,2}&=\pm\frac{R}{2L}+\sqrt{\left(\frac{R}{2L}\right)^{2}+\frac{1}{LC}}\\
		\omega_{c1}\omega_{c2}&=\omega_{0}^{2}\\
		\therefore\omega_{0}&=\sqrt{\omega_{c1}\omega_{c2}}
	\end{align*}
	Bandwidth, $\beta$:
	\begin{align*}
		\beta&=|\omega_{c1}-\omega_{c2}|\\
		&=\frac{R}{L}
	\end{align*}
	Quality factor, Q, is the width of the bandpass independent of the center frequency.
	\begin{align*}
		Q&=\frac{\omega_{0}}{\beta}=\frac{\sqrt{\frac{1}{LC}}}{\frac{R}{L}}\\
		&=\frac{1}{R}\sqrt{\frac{L}{C}}\\
		H(j\omega)&=\frac{j\omega\frac{R}{L}}{-\omega^{2}+j\omega\frac{R}{L}+\frac{1}{LC}}\\
		H(s)&=\frac{s\frac{R}{L}}{s^{2}+s\frac{R}{L}+\frac{1}{LC}}\\
		\omega_{0}&=\sqrt{\frac{1}{LC}}\\
		\beta&=\frac{R}{L}\\
		H(s)&=\frac{s\beta}{s^{2}+s\beta+\omega_{0}^{2}}
	\end{align*}
	\subsubsection{Band-reject Filter}
	\begin{center}
		\includegraphics[scale=0.5]{passive-bandreject-filter.png}
	\end{center}
	\begin{align*}
		H(j\omega)&=\frac{j\omega L+\frac{1}{j\omega C}}{R+j\omega L+\frac{1}{j\omega C}}\\
		&=\frac{-\omega^{2}+\frac{1}{LC}}{-\omega^{2}+j\omega\frac{R}{L}+\frac{1}{LC}}
	\end{align*}
	Magnitude Response:
	$$|H(j\omega)|=\frac{|\frac{1}{LC}-\omega^{2}|}{\sqrt{\left(\frac{1}{LC}-\omega^{2}\right)^{2}+\left(\frac{\omega R}{L}\right)^{2}}}$$
	Phase Response:
	\begin{align*}
		\angle H(j\omega)&=\arctan\left(\frac{\frac{-\omega R}{L}}{\frac{1}{LC}-\omega^{2}}\right)\\
		&=-\arctan\left(\frac{\frac{\omega R}{L}}{\frac{1}{LC}-\omega^{2}}\right)
	\end{align*}
	Cut off Frequency:
	$$\omega_{c1,2}=\pm\frac{R}{2L}+\sqrt{\left(\frac{R}{2L}\right)^{2}+\left(\frac{1}{LC}\right)}$$
	Center Frequency:
	$$\omega_{c}=\sqrt{\frac{1}{LC}}$$
	Bandwidth:
	$$\beta=\frac{R}{L}$$
	Q Factor:
	$$Q=\frac{1}{R}\sqrt{\frac{L}{C}}$$
	Transfer Function:
	\begin{align*}
		H(j\omega)&=\frac{-\omega^{2}+\frac{1}{LC}}{-\omega^{2}+j\omega\frac{R}{L}+\frac{1}{LC}}\\
		H(s)&=\frac{s+\frac{1}{LC}}{s^{2}+s\frac{R}{L}+\frac{1}{LC}}\\\\
		\omega_{0}&=\sqrt{\frac{1}{LC}}\\
		\beta&=\frac{R}{L}\\
		H(s)&=\frac{s+\omega_{0}^{2}}{s+s\beta+\omega_{0}^{2}}
	\end{align*}
